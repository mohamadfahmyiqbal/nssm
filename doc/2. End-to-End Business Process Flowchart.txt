        B3 --> B4[Poller Workers Dequeue Jobs secara Paralel]
        
        B4 --> B5{Vendor Hardware}
        B5 -- "Cisco / HPE" --> B6[Query SNMP OID Port, CPU, Uptime]
        B5 -- "UPS System" --> B7[Query SNMP OID Battery & AC Power]
        B5 -- "Panasonic NVR" --> B8[Query ICMP Ping / Availability]

        B12 -- "Tidak" --> B10
        B12 -- "Ya" --> B13["Set Temp Status = DOWN"]
    end

    %% FASE 3: STATUS EVALUATION & ALERTING
    subgraph Phase3 ["FASE 3: Evaluasi Status & Email Alert Gateway"]
        B10 & B13 --> C1[Evaluator Engine Cek Status Sebelumnya di DB]
        C1 --> C2[Update Record Status & LastPolledAt di SQL Server]
        C2 --> C3["Emit Push Event Socket.io 'device:status_update'"]
        
        C1 --> C4{Status Berubah ke DOWN?}

        C6 -- "Clear" --> C8[Trigger Nodemailer SMTP - Kirim Email Alert]
        C8 --> C9["Set Redis Cooldown Flag = 15 Menit"]
    end
