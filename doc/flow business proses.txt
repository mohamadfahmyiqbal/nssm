flowchart TD
    %% ===================================================
    %% FASE 1: AUTHENTICATION & INITIALIZATION
    %% ===================================================
    subgraph Phase1 ["1. Fase Otentikasi & Inisialisasi Sesi"]
        A1([Mulai App]) --> A2[User Mengakses Web Dashboard]
        A2 --> A3[Input Username & Password]
        A3 --> A4{Validasi Credential via REST API}
        
        A4 -- "Gagal (401/403)" --> A5[Tampilkan Pesan Error / Akses Ditolak] --> A3
        A4 -- "Sukses (200)" --> A6[Generate JWT Token & Simpan di LocalStorage]
        A6 --> A7[Inisialisasi Koneksi WebSocket Socket.io + Header JWT]
        A7 --> A8[Fetch Data Perangkat, Nodes & Links dari SQL Server]
        A8 --> A9[Render Canvas Peta Topologi & Status Summary Bar]
    end

    %% ===================================================
    %% FASE 2: BACKGROUND QUEUE & POLLING ENGINE
    %% ===================================================
    subgraph Phase2 ["2. Siklus Background Polling & Queuing (BullMQ/Redis)"]
        B1([Scheduler Cron Tick - Interval 30 Detik]) --> B2[Query Perangkat Aktif dari SQL Server]
        B2 --> B3[Enqueue Jobs Per-Device ke BullMQ / Memurai]
        B3 --> B4[Poller Workers Dequeue Jobs secara Paralel]
        
        B4 --> B5{Tipe Vendor Perangkat}
        B5 -- "Cisco / HPE" --> B6[Query SNMP v2c/v3 OID CPU, Port Status]
        B5 -- "UPS Infrastructure" --> B7[Query SNMP OID Battery & AC Power]
        B5 -- "Panasonic NVR" --> B8[Query ICMP Ping / SNMP Availability]
        
        B6 & B7 & B8 --> B9{Apakah Perangkat Merespon?}
        
        B9 -- "Ya (Responsive)" --> B10[Set Temp Status = UP / WARNING]
        B9 -- "Tidak (Timeout)" --> B11[Retry Mechanism - Max 2x] --> B12{Tetap Timeout?}
        
        B12 -- "Tidak" --> B10
        B12 -- "Ya" --> B13[Set Temp Status = DOWN]
    end

    %% ===================================================
    %% FASE 3: STATUS EVALUATION & EMAIL ALERT
    %% ===================================================
    subgraph Phase3 ["3. Evaluasi Status & Email Alert Gateway"]
        B10 & B13 --> C1[Evaluator Engine Bandingkan dengan Status di DB]
        C1 --> C2[Update Record CurrentStatus & LastPolledAt di SQL Server]
        C2 --> C3[Emit Push Event WebSocket 'device:status_update' ke Frontend]
        
        C1 --> C4{Apakah Status Berubah ke DOWN?}
        C4 -- "Tidak (Tetap/Up)" --> C5[Abaikan Alert]
        C4 -- "Ya (Transisi UP -> DOWN)" --> C6{Cek Redis Cooldown Flag (15 Menit)}
        
        C6 -- "Cooldown Active" --> C7[Abaikan - Mencegah Flooding Email]
        C6 -- "Cooldown Clear" --> C8[Trigger SMTP Nodemailer - Kirim Email Alert]
        C8 --> C9[Set Redis Cooldown Flag = 15 Menit]
    end

    %% ===================================================
    %% FASE 4: USER INTERACTION & REAL-TIME DASHBOARD
    %% ===================================================
    subgraph Phase4 ["4. Interaksi User & Dashboard Peta Real-Time"]
        C3 --> D1[Socket.io Listener Menerima Payload Baru]
        D1 --> D2[Ikon Node di Canvas Berubah Warna Instan<br/>(Hijau = UP | Kuning = WARNING | Merah = DOWN)]
        
        A9 --> D3[User Mengklik Node Perangkat di Canvas]
        D3 --> D4[Tampilkan Modal Detail: IP, CPU Usage, Uptime, Alert Log]
        
        A9 --> D5[User Menggeser Ikon Node (Drag and Drop)]
        D5 --> D6[Kirim Koordinat Baru PosX, PosY via REST API]
        D6 --> D7[Update Koordinat Node di SQL Server]
    end

    %% ===================================================
    %% FASE 5: DEVICE MANAGEMENT & AUTO-DISCOVERY
    %% ===================================================
    subgraph Phase5 ["5. Manajemen Perangkat & Auto-Discovery Topologi"]
        E1([User Klik 'Tambah Perangkat']) --> E2[Input Form: Hostname, IP, Vendor, SNMP Key]
        E2 --> E3[Simpan Record Perangkat Baru ke SQL Server]
        E3 --> E4[Trigger Engine Auto-Discovery LLDP / CDP]
        
        E4 --> E5[Query SNMP OID LLDP/CDP Neighbor ID & Remote Port]
        E5 --> E6{Apakah Ditemukan Switch Tetangga?}
        
        E6 -- "Ya (Neighbor Found)" --> E7[Simpan Hubungan Link ke Tabel TopologyLinks]
        E6 -- "Tidak (Standalone)" --> E8[Buat Node Tanpa Garis Hubungan]
        
        E7 & E8 --> E9[Re-render Canvas Peta Topologi]
    end
